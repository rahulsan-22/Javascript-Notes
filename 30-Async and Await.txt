//! Async & Await
--> Async and Await are ES8 (ES2017) features.
--> They provide a cleaner and more readable way to work with Promises.
--> Async/Await is built on top of Promises.
--> It helps avoid long Promise chaining using then() and catch().

//! async Keyword
--> async keyword is used before a function.
--> An async function always returns a Promise.
--> Even if you return a normal value, JavaScript automatically wraps it inside a Promise.

//*Example:
async function demo(){
   return "Hello";
}

console.log(demo());

//*Output:
Promise { "Hello" }

//! Accessing Returned Value
async function demo(){
   return "Hello";
}

demo()
.then((data)=>{
   console.log(data);
});

//*Output:
Hello

//! await Keyword
--> await keyword is used inside async functions only.
--> It pauses the execution of the async function until the Promise is resolved or rejected.
--> It does NOT block the entire JavaScript program.
--> It only pauses the current async function.

//*Syntax:
await promise;

//! Example Without await
function getData(){
   return new Promise((resolve)=>{
      setTimeout(()=>{
         resolve("Data Received");
      },2000);
   });

}
getData()
.then((data)=>{
   console.log(data);
});

//*Output After 2 Seconds:
Data Received

//! Example Using await
function getData(){
   return new Promise((resolve)=>{
      setTimeout(()=>{
         resolve("Data Received");
      },2000);
   });
}

async function fetchData(){
   let result = await getData();
   console.log(result);
}
fetchData();

//*Output After 2 Seconds:
Data Received

//! Error Handling Using try...catch

--> Used to handle rejected Promises.
--> Similar to catch() in Promise chaining.

//*Example:


//! try Block
--> Contains code that may throw an error.

//! catch Block
--> Executes if an error occurs inside try block.

//! finally Block
--> Executes whether error occurs or not.

//*Example:
try{
   console.log("Try Block");
}
catch(error){
   console.log(error);
}
finally{
   console.log("Always Executes");
}

//! Multiple Await Example
function task1(){
   return new Promise((resolve)=>{
      setTimeout(()=>{
         resolve("Task 1 Completed");
      },1000);
   });
}

function task2(){
   return new Promise((resolve)=>{
      setTimeout(()=>{
         resolve("Task 2 Completed");
      },2000);
   });
}

async function executeTasks(){
   let result1 = await task1();
   console.log(result1);
   let result2 = await task2();
   console.log(result2);
}
executeTasks();

//*Output:
Task 1 Completed
Task 2 Completed
