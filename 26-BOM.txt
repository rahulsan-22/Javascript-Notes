//! Browser Object Model (BOM)
--> BOM stands for Browser Object Model.
--> BOM allows JavaScript to interact with the Browser.
--> Using BOM, we can control browser windows, navigate pages, display dialogs, get browser information, and much more.
--> BOM is provided by the Browser, not by JavaScript itself.

//! BOM Hierarchy
window
│
├── document
├── location
├── history
├── navigator
├── screen
└── console

//! Note:
--> window is the top-level object which represents the browser window.
--> All BOM objects are properties of the window object.

//*Example
console.log(window);

//! Window Methods

//!alert()
--> Displays a message box with an OK button.
//*Example:
alert("Welcome to JavaScript");

//! confirm()
--> Displays a confirmation box with OK and Cancel buttons.
--> Returns true or false.
//*Example:
let result = confirm("Are you sure?");
console.log(result);
//*Note:
OK     => true
Cancel => false

//! prompt()
--> Displays an input box.
--> Returns user input.
//*Example:
let name = prompt("Enter Your Name");
console.log(name);

//! open()
--> Opens a new browser window/tab.
//*Example:
open("https://www.google.com");

//! close()
--> Closes the current browser window.
//*Example:
close();

//! print()
--> Opens the browser print dialog.
//*Example:
window.print();

//! Window Properties

//!innerWidth
--> Returns browser viewport width.
//*Example:
console.log(window.innerWidth);

//!innerHeight
--> Returns browser viewport height.
//*Example
console.log(window.innerHeight);

//! Location Object
--> location object contains information about the current URL.
--> Used for navigation and URL manipulation.

//*Example
console.log(location);

//! Common Location Properties

//!href
--> Returns complete URL.
console.log(location.href);
//*Example: https://www.google.com/search?q=javascript

//!protocol
--> Returns protocol.
console.log(location.protocol);

//*Example: https:

//!hostname
--> Returns domain name.
console.log(location.hostname);
//*Example: www.google.com

//!pathname
--> Returns page path.
console.log(location.pathname);

//*Example: /search

//! Location Methods

//!reload()
--> Reloads the current page.
//*Example:
location.reload();

//! assign()
--> Navigates to another page.
//*Example:
location.assign("https://www.google.com");

//! replace()
--> Replaces current page.
--> User cannot return using Back button.
//*Example:
location.replace("https://www.google.com");

//! History Object
--> Used to navigate browser history.
//*Example
console.log(history);

//! History Methods

//!back()
--> Moves to previous page.
//*Example:
history.back();

//! forward()
--> Moves to next page.
//*Example:
history.forward();

//! go()
--> Moves to a specific history position.
history.go(-2); //2 pages back
history.go(3);  //3 pages forward

//! Navigator Object
--> Provides information about the browser.
//*Example
console.log(navigator);

//! Common Navigator Properties

//!language
--> Returns browser language.
//*Example:
console.log(navigator.language); // en-US

//!onLine
--> Checks internet connection status.
//*Example:
console.log(navigator.onLine);
//*Note:
true  => Online
false => Offline

//! Geolocation API
--> Used to get the user's current location.
--> Available through navigator.geolocation.
--> Requires user permission.

//*Syntax
navigator.geolocation.getCurrentPosition(
   (position) => {
      console.log(position.coords.latitude);
      console.log(position.coords.longitude);
   }
);

//! Web Storage API (localStorage & sessionStorage)
--> Web Storage API allows storing data in the browser.
--> Data is stored as Key-Value pairs.
--> Both store data as Strings.
--> If you store Arrays or Objects, convert them using JSON.stringify().

//! Types of Storage
1. localStorage
2. sessionStorage

//! localStorage
--> Stores data permanently until manually removed.
--> Data remains even after closing the browser.
--> Shared across tabs of the same origin.

//*Store Data
localStorage.setItem("name", "Rahul");

//*Retrieve Data
console.log(
   localStorage.getItem("name")
);

//*Remove Specific Item
localStorage.removeItem("name");

//*Remove All Items
localStorage.clear();

//! sessionStorage
--> Stores data only for the current browser session.
--> Data is removed when the tab/browser is closed.
--> Not shared between tabs.

//*Store Data
sessionStorage.setItem("username","Rahul");

//*Retrieve Data
console.log(
   sessionStorage.getItem("username")
);

//*Remove Specific Item
sessionStorage.removeItem(
   "username"
);

//*Remove All Items
sessionStorage.clear();
