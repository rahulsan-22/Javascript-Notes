//!Arrays
--> Array is a special object used to store multiple values in a single variable.
--> Arrays can store different datatypes
--> Arrays are mutable
--> Arrays are dynamic in size
//*Example:
let arr = [10, 20, 30];
let data = ["Rahul", 25, true, null];

//!Ways to Create Arrays
//*1) Using Array Literal
let fruits = ["Apple", "Mango", "Orange"];
console.log(fruits) //['Apple', 'Mango', 'Orange']
console.log(typeof fruits) //object

//*2) Using Array Constructor
let nums = new Array(10, 20, 30);
console.log(nums) //[10,20,30]
console.log(typeof nums) //object

//!Accessing Elements in Arrays
-->Using Indexing
let colors = ["Red", "Blue", "Green"];
console.log(colors[0]); //Red
console.log(colors[1]); //Blue
console.log(colors[10]); //undefined
console.log(colors[-1]); //undefined

//*Note:
--> Array indexing starts from 0.
--> colors[0] means first element.
--> If index does not exist, JavaScript returns undefined.
--> JavaScript arrays DO NOT support negative indexing using [] returns undefined.

//!Length Property:
--> length returns total number of elements
let arr = ["Html","Css","Js"];
console.log(arr.length); //3
console.log(arr[arr.length - 1]); //Js
console.log(arr[arr.length - 2]); //Css
//*Note:
-->length - 1 always gives last index.

//!Arrays are mutable (Values can be changed after creation)
let fruits = ["Apple", "Mango"];
fruits[1] = "Orange";
console.log(fruits); //["Apple", "Orange"]

//!Methods for Adding Elements

//*1)push()
--> Adds element at end of an array
let arr = [10, 20];
arr.push(30);
console.log(arr); //[10,20,30]

//*2)unshift()
--> Adds element at beginning of an array
let arr = [20, 30];
arr.unshift(10);
console.log(arr); //[10,20,30] 

//!Methods for Removing Elements

//*1)pop()
--> Removes last element from an array
let arr = [10, 20, 30];
arr.pop();
console.log(arr); //[10,20]

//*2)shift()
--> Removes first element from an array
let arr = [10, 20, 30];
arr.shift();
console.log(arr); //[20,30]

//!Methods

//!1) Array.isArray()
--> Used to check whether a value is an array or not.
--> Returns true if the value is an array.
--> Returns false for any other datatype.
--> Commonly used for validation before applying array methods.
//*Example:
console.log(Array.isArray([10,20])) //true
console.log(Array.isArray("Rahul")) //false
console.log(Array.isArray([])) //true
console.log(Array.isArray(new Array(1,2,3))) //true

//!2) toString()
--> Converts all array elements into a single string.
--> Elements are separated by commas (,).
--> Original array remains unchanged.
--> Return type is string.
//*Example:
let arr2 = ["Red","Green","Blue"]
console.log(arr2.toString()) //Red,Green,Blue
console.log(typeof arr2.toString()) //string

//!3) concat(...arr)
--> Used to merge two or more arrays.
--> Does not modify the original arrays.
--> Returns a new array.
--> Can also add normal values while concatenating.
//*Example:
let sub1 = ["Html","Css"]
let sub2 = ["Js"]
console.log(sub1.concat(sub2)) //['Html', 'Css', 'Js']
console.log(sub2.concat(sub1,"React")) //['Js', 'Html', 'Css', 'React']

//!4) join(separator)
--> Converts array elements into a string.
--> Custom separator can be provided.
--> Original array remains unchanged.
--> If no separator is provided, comma (,) is used by default.
//*Example:
let arr4 = ["rcb","srh","csk","dc"]
console.log(arr4.join("☠️")) //rcb☠️srh☠️csk☠️dc
console.log(arr4.join("-")) //rcb-srh-csk-dc

//!5) reverse()
--> Reverses the order of elements in an array.
--> Modifies the original array.
--> Returns the reversed array.
--> It is a mutable method.
//*Example:
let arr5 = ["Marker","Laptop","Remote"]
console.log(arr5.reverse()) //['Remote', 'Laptop', 'Marker']

//!6) includes(searchElement,startIndex)
--> Checks whether an element exists in an array.
--> Returns true if found.
--> Returns false if not found.
--> Search starts from startIndex if provided.
--> Supports negative indexing for startIndex.
//*Example:
let arr6 = ["Html","Css","Js","React","Java","Css","Sql"]
console.log(arr6.includes("Js")) //true
console.log(arr6.includes("Python")) //false
console.log(arr6.includes("Js",4)) //false
console.log(arr6.includes("Css",3)) //true
console.log(arr6.includes("React",-5)) //true
console.log(arr6.includes("Java",-500)) //true

//!7) indexOf(searchElement,startIndex)
--> Returns the index of the first matching element.
--> Returns -1 if the element is not found.
--> Search starts from startIndex if provided.
--> Supports negative indexing.
--> Useful for finding positions of elements.
//*Example:
let arr7 = ["Html","Css","Js","React","Java","Css","Sql"]
console.log(arr7.indexOf("Js")) //2
console.log(arr7.indexOf("Python")) //-1
console.log(arr7.indexOf("Js",4)) //-1
console.log(arr7.indexOf("Css",3)) //5
console.log(arr7.indexOf("Css",1)) //1
console.log(arr7.indexOf("React",-5)) //3
console.log(arr7.indexOf("Java",-500)) //4

//!8) slice(startIndex,endIndex)
--> Used to extract a portion of an array.
--> Start index is inclusive.
--> End index is exclusive.
--> Does not modify the original array.
--> Supports negative indexing.
--> Returns a new array.
//*Example:
let arr8 = ["Maroon","Red","Green","Blue","Yellow"]
console.log(arr8.slice(2)) //['Green', 'Blue', 'Yellow']
console.log(arr8.slice(1,4)) //['Red', 'Green', 'Blue']
console.log(arr8.slice(2,3)) //['Green']
console.log(arr8.slice(0,0)) //[]
console.log(arr8.slice(-4,-2)) //['Red', 'Green']
console.log(arr8.slice(-4,4)) //['Red', 'Green','Blue']
console.log(arr8.slice(-4,-5)) //[]
console.log(arr8.slice(5,2)) //[]

//*Note:
--> Original array remains unchanged.
--> If startIndex > endIndex, returns empty array.
--> Commonly used for copying arrays.


//!9) splice(startIndex,deleteCount,elementsToBeAdded)
--> Used to add, remove, or replace elements.
--> Modifies the original array.
--> Start index specifies where operation begins.
--> deleteCount specifies number of elements to remove.
--> Supports negative indexing.
--> It is a mutable method.
//*Example:
let arr9 = ["Kingfisher","Budwiser","OC"]

//*Insert Elements
arr9.splice(1,0,"Tuborg","Corona")
console.log(arr9) //['Kingfisher', 'Tuborg', 'Corona', 'Budwiser', 'OC']

//*Delete Elements
arr9.splice(3,2)
console.log(arr9) //['Kingfisher', 'Tuborg', 'Corona']

//*Replace Elements
arr9.splice(1,2,"MansionHouse","BroCode")
console.log(arr9) //['Kingfisher', 'MansionHouse', 'BroCode']

//*Negative Indexing
arr9.splice(-3,2,"RedLabel","BlueLabel","BlackLabel")
console.log(arr9) //['RedLabel', 'BlueLabel', 'BlackLabel', 'BroCode']


//!slice() vs splice()

//*slice()
--> Original array remains unchanged.
--> Used to extract elements.
--> Returns extracted portion.

//*splice()
--> Original array gets modified.
--> Used to insert, delete, and replace elements.
--> Returns deleted elements.


//!Higher Order Methods
--> Higher Order Methods accept callback functions as arguments.
--> They reduce the need for traditional loops.
--> Widely used in modern JavaScript and React applications.


//!1) forEach(fn(element,index,array))
--> Executes callback for every element.
--> Used for iteration.
--> Does not return anything.
--> Return value is always undefined.
--> Cannot create a new array.
//*Example:
let trainers = ["Monty","Pavan"]
trainers.forEach((element,index,array)=>{
   console.log(element+"🔥") //Monty🔥 //Pavan🔥
   console.log(element.length) //5 //5
   console.log(element.toUpperCase()) //MONTY //PAVAN
   console.log(element.at(0)) //M //P
   console.log(index) //0 //1
   console.log(array) //["Monty","Pavan"] * 2
})


//!2) map(fn(element,index,array))
--> Executes callback for every element.
--> Returns a new transformed array.
--> Original array remains unchanged.
--> Most commonly used for data transformation.
//*Example:
trainers.map((element,index,array)=>{
   console.log(element+"🔥") //Monty🔥 //Pavan🔥
   console.log(element.length) //5 //5
   console.log(element.toUpperCase()) //MONTY //PAVAN
   console.log(element.at(0)) //M //P
   console.log(index) //0 //1
   console.log(array) //["Monty","Pavan"] * 2
 })


//*forEach() vs map()
let num = [1,2,3]

--> Return value is always undefined.
let forEachOutput = num.forEach((ele)=>{
  return ele*2
})
console.log(forEachOutput) //undefined

--> Returns a new transformed array.
let mapOutput = num.map((ele)=>{
   return ele*2
})
console.log(mapOutput) //[2,4,6]

//!3) filter(fn(element,index,array))
--> Returns all elements that satisfy a condition.
--> Returns a new array.
--> Original array remains unchanged.
--> If no element matches, returns empty array.
//*Example:
let marks = [65,35,80,45,49]
let filteredOutput = marks.filter((ele)=>{
   return ele>45
})
console.log(filteredOutput) //[65,80,49]


//!4) find(fn(element,index,array))
--> Returns the first matching element.
--> Stops searching after first match.
--> Returns undefined if no match is found.
--> Does not return an array.
//*Example:
let findOutput = marks.find((ele)=>{
   return ele>55
 })
console.log(findOutput) //65


//!find() vs filter()

//*find()
--> Returns first matching element.
--> Stops after first match.
--> Return type is element or undefined.

//*filter()
--> Returns all matching elements.
--> Searches entire array.
--> Return type is array.


//!5) some(fn(element,index,array))
--> Returns true if at least one element satisfies the condition.
--> Returns false if no element satisfies the condition.
--> Stops execution once condition becomes true.
//*Example:
let someOutput = marks.some((ele)=>{
  return ele>45
})
console.log(someOutput) //true


//!6) every(fn(element,index,array))
--> Returns true only if all elements satisfy the condition.
--> Returns false if any one element fails.
--> Stops execution once condition becomes false.
//*Example:
let everyOutput = marks.every((ele)=>{
  return ele>45
})
console.log(everyOutput) //false


//!some() vs every()

//*some()
--> At least one element must satisfy the condition.
--> Logical OR behavior.

//*every()
--> All elements must satisfy the condition.
--> Logical AND behavior.

//!7) reduce(fn(accumulator,element,index,array),initialValue)
--> Used to reduce an array into a single value.
--> Can perform sum, product, average, counting, grouping etc.
--> Accumulator stores the previous result.
--> Initial value is optional but recommended.
//*Example 1
let arr7 = [10, 20, 30, 40, 50]
let total = arr7.reduce((accumulator,element,index,array) => {
   return accumulator+=element
}, 0)
console.log(total) //150

//*Example 2
let result = arr7.reduce((sum,ele) => {
   return sum+=ele
}, 0)
let avg = result / arr7.length
console.log(avg) //30

//!8) sort(compareFn)
--> Used to sort array elements.
--> Modifies the original array.
--> By default converts values into strings.
--> Performs lexicographical (dictionary order) sorting by default.
--> For numbers always use compare function.
//*Note
a-b => Ascending Order
b-a => Descending Order
--> If compareFunction returns a negative value, 'a' comes before 'b'.
--> If compareFunction returns a positive value, 'b' comes before 'a'.
--> If compareFunction returns 0, positions remain unchanged.

//*Example 1:
let num = [7, 2, 1, 4, 5]
console.log(num.sort()) //[1,2,4,5,7]

//*Example 2:
let marks = [1, 10, 5, 25, 2]
console.log(marks.sort((a,b)=>a-b)) //[1, 2, 5, 10, 25]
console.log(marks.sort((a,b)=>b-a)) //[25, 10, 5, 2, 1]

//*Example 3:
let random = ["Banana", "Orange", true ,"Apple","Apricot"]
console.log(random.sort()) //['Apple', 'Apricot', 'Banana', 'Orange', true]
