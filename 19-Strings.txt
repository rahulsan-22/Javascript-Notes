//!STRINGS:
--> A String is a sequence of characters used to represent text in JavaScript.
--> Strings are immutable in JavaScript (original string cannot be changed).
//*Examples:
let name = "Rahul";
let city = 'Hyderabad';
let message = `Welcome`;

//!Ways to Create Strings:

//*1) Using Single Quotes
let str1 = 'Something'
console.log(str1) //Something
console.log(typeof str1) //string

//*2) Using Double Quotes
let str2 = "Breakfast"
console.log(str2) //Breakfast
console.log(typeof str2) //string

//*3) Using Template Literals (Backticks)
--> Used for interpolation and multi-line strings.
//@Example:
let str3 = `Pavan`
console.log(`My name is ${str3} and my age is ${20 + 5 + 1}`)
console.log(`My 
name 
is 
Pavan`)
console.log(str3) //Pavan
console.log(typeof str3) //string
Note:
--> ${} is called interpolation.
--> It allows inserting variables or expressions inside strings..

//!Accessing Characters in String:
-->Using Index:
let subject = "JAVASCRIPT"
console.log(subject[0]) //J
console.log(subject[4]) //S
console.log(subject[10]) //undefined
console.log(subject[-1]) //undefined

//*Note:
--> String indexing starts from 0.
--> str[0] means first character.
--> If index does not exist, JavaScript returns undefined.
--> JavaScript strings DO NOT support negative indexing using [] returns undefined.

//!Length Property:
--> length returns total number of characters including spaces.
let str = "JavaScript";
console.log(str.length); //10
console.log(str[str.length - 1]); //t
console.log(str[str.length - 2]); //p
//*Note:
-->length - 1 always gives last index.

//!BUILT-IN METHODS:

//!1) toUpperCase()
--> Converts string into uppercase.
//*Example:
let str1 = "MaRkER"
console.log(str1.toUpperCase()) //MARKER

//!2) toLowerCase()
--> Converts string into lowercase.
//*Example:
let str2 = "ReMoTE"
console.log(str2.toLowerCase()) //remote

//!3) trim()
--> Removes spaces from beginning and ending.
//*Example:
let str3 = "         hello    everyone      "
console.log(str3) //         hello    everyone      
console.log(str3.length) //32
console.log(str3.trim()) //hello    everyone
console.log((str3.trim()).length) //17
//*Note:
--> Does not remove spaces between words.

//!4) at(index)
--> Returns character at specified index.
--> Supports negative indexing.
//*Example:
let str4 = "JAVASCRIPT"
console.log(str4.at(0)) //J
console.log(str4.at(7)) //I
console.log(str4.at(10)) //undefined
console.log(str4.at(-1)) //T

//!5) slice(startIndex, endIndex)
--> Extracts part of string.
--> Supports Negative Indexing: values start from end.
--> If StartIndex > EndIndex returns empty string.
//*Example:
let str5 = "GOOD MORNING"
console.log(str5.slice(3)) //D MORNING
console.log(str5.slice(0, 3)) //GOO
console.log(str5.slice(6, 8)) //OR
console.log(str5.slice(-4)) //NING
console.log(str5.slice(-7, -2)) //MORNI
console.log(str5.slice(3, -3)) //D MORN
console.log(str5.slice(-1, -4)) //""
console.log(str5.slice(-10, 7)) //OD MO
console.log(str5.slice(10,2)) //""
console.log(str5.slice(100, 2)) //""
console.log(str5) //GOOD MORNING
//*Note:
--> Start index included whereas End index excluded.
--> If only Start Index specified extracts from that start index till end.
--> Original string does not change.

//!6) substring(startIndex, endIndex)
--> Similar to slice() but does not support negative indexing.
--> Negative index values become 0.
--> If StartIndex > EndIndex swaps values.
//*Example:
let str6 = "GOOD MORNING"
console.log(str6.substring(3)) //D MORNING
console.log(str6.substring(0, 3)) //GOO
console.log(str6.substring(6, 8)) //OR
console.log(str6.substring(-4)) //GOOD MORNING
console.log(str6.substring(-7, -2)) //""
console.log(str6.substring(3, -3)) //GOO
console.log(str6.substring(-1, -4)) //""
console.log(str6.substring(-10, 7)) //GOOD MO
console.log(str6.substring(10,2)) //OD MORNI
console.log(str6.substring(100, 2)) //OD MORNING
console.log(str6) //GOOD MORNING

//!7) includes(searchValue, startIndex)
--> Checks whether substring exists.
--> Negative index values become 0.
--> Returns boolean value i.e true if exists and false if not exists
//*Example:
let str7 = "PAVAN SIR"
console.log(str7.includes("A")) //true
console.log(str7.includes("Z")) //false
console.log(str7.includes("A",1)) //true
console.log(str7.includes("A", 4)) //false
console.log(str7.includes("VA")) //true
console.log(str7.includes("VA", 3)) //false
console.log(str7.includes("SIR",-7)) //true
console.log(str7.includes("SIR",-2)) //true

//!8) indexOf(searchValue, startIndex)
--> Returns first occurrence index.
--> Negative index values become 0.
--> If value not found returns -1.
//*Example:
let str8 = "PAVAN SIR"
console.log(str8.indexOf("A")) //1
console.log(str8.indexOf("Z")) //-1
console.log(str8.indexOf("A",1)) //1
console.log(str8.indexOf("A", 4)) //-1
console.log(str8.indexOf("VA")) //2
console.log(str8.indexOf("VA", 3)) //-1
console.log(str8.indexOf("SIR",-7)) //6
console.log(str8.indexOf("SIR", -2)) //6

//!9) replace(oldValue, newValue)
--> Replace Only First Occurrence.
//*Example:
let str9 = "BIRYANI BIRYANI BIRYANI"
console.log(str9.replace("BIRYANI","SHAWARMA")) //SHAWARMA BIRYANI BIRYANI
console.log(str9.replaceAll("BIRYANI","SHAWARMA")) //SHAWARMA SHAWARMA SHAWARMA
//*Note:
--> To replace all occurrences use replaceAll().

//!10) split(separator, limit)
--> Converts string into array by splitting it based on the separator.
//*Example:
let str10 = "RCB GT RR"
console.log(str10.split(" ")) //['RCB','GT','RR']
console.log(str10.split(" ", 1)) //['RCB']
console.log(str10.split("-")) //['RCB GT RR']
console.log(str10.split("")) //['R', 'C', 'B', ' ', 'G', 'T', ' ', 'R', 'R']
console.log(str10.split("R")) //['','CB GT ','','']

//!11) startsWith(searchValue, startIndex)
--> Checks whether string starts with specified value.
//*Example:
let str11 = "SOMETHING"
console.log(str11.startsWith("S")) //true
console.log(str11.startsWith("O")) //false
console.log(str11.startsWith("SO")) //true
console.log(str11.startsWith("SOP")) //false
console.log(str11.startsWith("SO",4)) //false
console.log(str11.startsWith("SO",-4)) //true
console.log(str11.startsWith("SO",-100)) //true
console.log(str11.startsWith("TH", 4)) //true

//!12) endsWith(searchValue, length)
--> Checks whether string ends with specified value.
//*Example:
let str12 = "ANYTHING"
console.log(str12.endsWith("NG")) //true
console.log(str12.endsWith("NI")) //false
console.log(str12.endsWith("NG",8)) //true
console.log(str12.endsWith("TH",5)) //true
console.log(str12.endsWith("TH",-1)) //false
console.log(str12.endsWith("A",-10)) //false
console.log(str12.endsWith("",-100)) //true

//*Note:
--> String methods are case-sensitive.
--> [] does not support negative indexing.
